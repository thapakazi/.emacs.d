#+begin_src emacs-lisp
  ;; --- Dynamic project capture ---
  ;; Reads top-level headings from projects.org via completing-read (vertico).
  ;; Pick an existing project or type a new name to create one on the fly.
  (defun kazi/org-capture-project-heading ()
    "Move point to a project heading in projects.org, picked via completing-read.
  If the heading doesn't exist, create it."
    (let* ((file "~/org/agendas/projects.org")
           (headings (with-temp-buffer
                       (when (file-exists-p file)
                         (insert-file-contents file)
                         (org-mode)
                         (org-map-entries
                          (lambda () (org-get-heading t t t t))
                          "LEVEL=1"))))
           (choice (completing-read "Project: " headings nil nil)))
      (find-file file)
      (goto-char (point-min))
      (if (re-search-forward (format "^\\* %s" (regexp-quote choice)) nil t)
          (end-of-line)
        ;; New project — create it at the end of file
        (goto-char (point-max))
        (unless (bolp) (insert "\n"))
        (insert (format "* %s\n" choice))
        (backward-char))))

  ;; FAST capture - minimal friction
  (setq org-capture-templates
        '(
          ;; Quick dump - no questions asked
          ("d" "Dump (quick)" entry (file+headline "~/org/agendas/inbox.org" "Tasks")
           "* TODO %?\n:PROPERTIES:\n:CREATED: %U\n:END:"
           :empty-lines 1)

          ;; Schedule for today
          ("t" "Today" entry (file "~/org/agendas/life.org")
           "* TODO %?\nSCHEDULED: %t"
           :empty-lines 1)

          ;; Scheduled task - asks for date
          ("s" "Scheduled" entry (file "~/org/agendas/life.org")
           "* TODO %?\nSCHEDULED: %^t"
           :empty-lines 1)

          ;; Deadline task
          ("D" "Deadline" entry (file "~/org/agendas/life.org")
           "* TODO %?\nDEADLINE: %^t"

  (defvar-local kazi/agenda-preview-mode nil
    "When non-nil, show a narrowed preview of the current agenda item.")

  (defun kazi/agenda-preview-update ()
    "Show a narrowed preview of the org entry at point in a side window."
    (when (and kazi/agenda-preview-mode
               (eq major-mode 'org-agenda-mode))
      (let ((marker (org-get-at-bol 'org-marker)))
        (if (not marker)
            (when-let ((win (get-buffer-window kazi/agenda-preview-buffer)))
              (with-current-buffer kazi/agenda-preview-buffer
                (let ((inhibit-read-only t)) (erase-buffer))))
          (let* ((buf (marker-buffer marker))
                 (pos (marker-position marker))
                 preview-text)
            (when buf
              (with-current-buffer buf
                (org-with-wide-buffer
                 (goto-char pos)
                 (let ((elt (org-element-at-point)))
                   (setq preview-text
                         (buffer-substring-no-properties
                          (org-element-property :begin elt)
                          (org-element-property :end elt))))))
              (with-current-buffer (get-buffer-create kazi/agenda-preview-buffer)
                (let ((inhibit-read-only t))
                  (erase-buffer)
                  (insert preview-text)
                  (org-mode)
                  (goto-char (point-min))
                  (org-fold-show-subtree)
                  (setq buffer-read-only t)))
              (display-buffer kazi/agenda-preview-buffer
                              '(display-buffer-in-side-window
                                (side . right)
                                (window-width . 0.4)
                                (preserve-size . (t . nil))))))))))

  (defun kazi/agenda-preview-toggle ()
    "Toggle narrowed preview for the current agenda item."
    (interactive)
    (setq kazi/agenda-preview-mode (not kazi/agenda-preview-mode))
    (if kazi/agenda-preview-mode
        (progn
          (add-hook 'post-command-hook #'kazi/agenda-preview-update nil t)
          (kazi/agenda-preview-update)
          (message "Agenda preview ON"))
      (remove-hook 'post-command-hook #'kazi/agenda-preview-update t)
      (when-let ((win (get-buffer-window kazi/agenda-preview-buffer)))
        (delete-window win))
      (message "Agenda preview OFF")))

    (defun kazi/agenda-focus--make-keys (headings)
      "Generate unique single-char keys from HEADINGS.
Uses first char of each word, then second char, etc."
      (let ((used (list ?|))  ;; reserve | for clear filter
            result)
        (dolist (h headings (nreverse result))
          (let* ((name (downcase h))
                 (key nil))
            (cl-block found
              ;; Try each char in the heading
              (dotimes (i (length name))
                (let ((ch (aref name i)))
                  (when (and (<= ?a ch) (<= ch ?z)
                             (not (member ch used)))
                    (setq key (char-to-string ch))
                    (push ch used)
                    (cl-return-from found))))
              ;; Fallback: first available letter
              (cl-loop for c from ?a to ?z
                       unless (member c used)
                       do (setq key (char-to-string c))
                          (push c used)
                          (cl-return-from found)))
            (push (cons key h) result)))))

* Cheatsheet                                                       :noexport:

** Quick Start: ~C-c a~ from anywhere, ~?~ inside agenda

*** Global: ~C-c a~ (or ~SPC o o~)
Opens transient with Capture / View / Actions columns.
Same keys as before: ~d~ dump, ~t~ today, ~w~ week, etc.

*** Inside Agenda: ~?~ (or ~C-o~)
Opens casual-agenda transient (extended with our custom groups):

| Section          | What                                            |
|------------------+-------------------------------------------------|
| Agenda           | ~d~ day, ~w~ week, ~m~ month, ~y~ year          |
| Filter           | ~/~ filter, ~\~ tag, ~=~ regexp, ~\vert~ clear   |
| *Quick Filter*   | ~F~ by tag, ~S~ by state, ~X~ clear all   |
| Actions          | ~o~ Operations, ~M~ Mark, ~k~ capture            |
| *Quick Schedule* | ~1~ today, ~2~ tomorrow, ~3~ next week, ~4~ search org |
| Navigation       | ~p/n~ lines, ~P/N~ items, ~b/f~ earlier/later    |
| Utils            | ~;~ timer, ~c~ calendar, ~l~ almanac              |
| Settings         | ~,~ settings (log, grid, diary, clock report)     |

** Daily Workflow

*** Morning (2 min)
1. ~C-c a t~ → See today's tasks
2. Pick one task, press ~t~ to mark as DOING

*** Throughout Day
- New thought? ~C-c a d~ → Dump it
- Know when to do it? ~C-c a s~ → Schedule it
- In agenda? Press ~?~ then ~s t~ to schedule today

*** Evening (5 min)
1. ~C-c a i~ → Review inbox
2. On each item: ~r~ to refile, ~?~ ~s s~ to schedule

** Weekly Review (Sunday, 15 min)

1. ~C-c w~ → Starts weekly review
2. Process inbox completely
3. ~C-c a w~ → See the week ahead
4. ~C-c a u~ → Check upcoming deadlines

** Inside Agenda View

| Key   | Action                           |
|-------+----------------------------------|
| ~?~   | *Casual transient (full menu)*   |
| ~t~   | Cycle TODO state                 |
| ~T~   | Schedule for TODAY               |
| ~M~   | Schedule for TOMORROW            |
| ~1~   | Quick: today (via transient)     |
| ~2~   | Quick: tomorrow (via transient)  |
| ~3~   | Quick: next week (via transient) |
| ~W~   | Filter: @work (via transient)    |

*** Navigation (Evil)

| Key   | Action              |
|-------+---------------------|
| ~j/k~ | Move down/up        |
| ~f/b~ | Forward/back by day |
| ~.~   | Go to today         |
